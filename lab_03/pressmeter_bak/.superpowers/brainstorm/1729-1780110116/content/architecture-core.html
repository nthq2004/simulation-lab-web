事件总线 (EventBus)

// 发布-订阅模式，单例
class EventBus {
  channels = {
    'equipment:select':    [subscribers],  // 设备选中
    'equipment:hover':     [subscribers],  // 设备悬停
    'equipment:stateChange': [subscribers], // 状态变更
    'equipment:alarm':     [subscribers],  // 报警触发
    'camera:focus':        [subscribers],  // 3D相机聚焦
    'view:switch':         [subscribers],  // 2D/3D/分屏切换
    'scene:load':          [subscribers],  // 加载场景
    'scene:reset':         [subscribers],  // 重置场景
    'workflow:step':       [subscribers],  // 教学步骤推进
  }
  emit(topic, payload) { ... }
  on(topic, callback) { ... }
  off(topic, callback) { ... }
}

// 2D → 3D 示例:
// 用户在原理图点击"1号发电机"
EventBus.emit('equipment:select', {id:'gen-01'})
// 3D 模块收到后高亮模型 + 聚焦相机

// 3D → 2D 示例:
// 3D 中某设备温度超限
EventBus.emit('equipment:alarm', {id:'me-01', type:'temp_high', value:98})
// 2D 原理图中该设备符号闪烁红色

设备对象模型

class EngineRoomEquipment {
  id: 'me-01'
  type: 'diesel_engine'
  label: '主机'

  // 2D/3D 引用 (由各渲染层注册)
  konvaRef: Konva.Group | null
  threeRef: THREE.Group | null

  // 传感器/执行器
  sensors: {
    rpm:    Sensor,
    temp_exhaust: Sensor,
    temp_coolant: Sensor,
    press_oil:    Sensor,
    fuel_flow:    Sensor,
  }
  actuators: {
    throttle: Actuator,
    start_air: Actuator,
  }

  // 状态
  state: {
    running: false,
    load_pct: 0,
    alarms: [],
  }

  // 物理模型引用
  physicsRef: Object | null
  // (指向求解器中的对应设备对象)
}

设备对象池

class EquipmentPool {
  devices: Map<string, EngineRoomEquipment>
  // 按系统分组便于查询
  systems: {
    'main_engine':  ['me-01'],
    'generators':   ['gen-01', 'gen-02'],
    'cooling':      ['pump-sw-01', 'pump-fw-01',
                     'hx-01', 'valve-01'],
    'fuel':         ['pump-fuel-01', 'purifier-01',
                     'tank-daily-01'],
    'air':          ['compressor-01', 'air-bottle-01'],
    'bilge':        ['pump-bilge-01'],
  }

  // 按系统查询所有设备
  getBySystem(system: string): EngineRoomEquipment[]
  // 查询某管路连通的所有设备
  getConnectedDevices(pipeId: string): EngineRoomEquipment[]
  // 广播状态变更
  notifyStateChange(devId: string, key: string, value: any)
}

物理仿真扩展

新增: ThermalSolver (热力求解器)

主机缸体/冷却水/排气/滑油的热交换计算：

  • 热节点网络 (热阻 + 热容模型)
  • 换热器模型 (板式/管壳式)
  • 燃烧放热模型 (基于负荷)
  • 与 PneumaticSolver 耦合 (冷却水流路)

扩展: 多介质管路系统

现有 PneumaticSolver 只支持单一介质（空气/水），扩展支持：

  • 燃油、滑油、淡水、海水、蒸汽
  • 每种介质独立物性参数 (密度/粘度/比热)
  • 多回路耦合 (如冷却水→换热器→海水)
  • 阀门/泵特性曲线

与现有 ControlSystem 的关系

关键设计决策: ControlSystem 保持为总控入口，新增模块作为其子模块扩展

// 在 consys.js 的 init() 中新增加载:
export class ControlSystem {
  init() {
    // 现有初始化...
    this.voltageSolver = new CircuitSolver(this)
    this.pressSolver   = new PneumaticSolver(this)
    this.digitalSolver = new DigitalSolver(this)
    // ...新增:
    this.thermalSolver = new ThermalSolver(this)  // 热力求解器
    this.equipmentPool = new EquipmentPool()       // 设备对象池
    this.eventBus      = new EventBus()             // 事件总线
    this.engineRoom3D  = null                       // 3D模块(按需加载)
  }

  _updatePhysics() {
    // 原顺序不变
    this.voltageSolver.update()
    this.pressSolver.solve()
    this.digitalSolver.update()
    // ...新增:
    this.thermalSolver?.solve()  // 热力计算
    this.equipmentPool?.syncInternalState() // 同步状态
    this._tickAll(dt)
  }
}
